// DATA STRUCTURES - It is used to group related variables together under one name

// Represents a single bike chosen by the customer.
// Stored in a vector so multiple bikes can be added per booking.
 
struct BikeSelection
{
    string typeName; // the name of the bike (e.g. "Standard")
    double typeCost; // base cost of that bike type
};

// Represents a single accessory added to a booking.
// Stored in a vector so multiple accessories can be added.

struct AccessorySelection
{
    string name;
    double cost;
};

// A complete, confirmed booking stored in session history.
// Using a struct groups all related data together cleanly, avoiding a long flat list of variables (good practice).
 
struct Booking
{
    vector<BikeSelection> bikes; // List of all bikes rented
    string durationName; // The rental duration chosen (e.g., "Full Day") (A customer can only pick one rental duration)
    double durationCost; // The cost associated with that duration
    vector<AccessorySelection> accessories; // A list of all accessories rented
    double totalCost; // The final calculated cost of the booking (bikes + duration + accessories)
};

/*
 * FUNCTION: getValidInt
 * Reads an integer from the user and makes sure it is valid.
 * The number must be between [minVal, maxVal].
 * It Shows a prompt message to the user.
 */
int getValidInt(const string& prompt, int minVal, int maxVal)
{
    int value; // stores the user’s input

    while (true) // repeat until valid input is given
    {
        cout << prompt; // show the prompt message
        cin >> value; // read input into 'value'

        if (cin.fail()) // case 1: user typed non-numeric input
        {
            cin.clear(); // reset input stream
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); // discard bad input
            cout << "[!] Invalid input - please enter a whole number between "
                << minVal << " and " << maxVal << "." << endl;
        }
        else if (value < minVal || value > maxVal) // case 2: number out of range
        {
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); 
            cout << "[!] Choice out of range - please enter a number between "
                << minVal << " and " << maxVal << "." << endl;
        }
        else // case 3: valid input
        {
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); 
            return value; // return the valid number
        }
    }
}

/*
 * FUNCTION: getValidPositiveDouble
 * Reads a decimal number (double) from the user and makes sure it is valid.
 * The number must be greater than zero (positive).
 */
double getValidPositiveDouble(const string& prompt)
{
    double value; // variable to store the user’s input

    while (true) // keep asking until valid input is given
    {
        cout << prompt; // show the prompt message
        cin >> value; // read input into 'value'

        if (cin.fail()) // CASE 1: user typed non-numeric input (e.g., "abc")
        {
            cin.clear(); // reset the input stream state
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); // discard leftover input
            cout << "[!] Invalid input - please enter a positive number." << endl;
        }
        else if (value <= 0.0) // CASE 2: user entered zero or a negative number
        {
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); 
            cout << "[!] Amount must be greater than zero." << endl;
        }
        else // CASE 3: input is valid (numeric and greater than zero)
        {
            cin.ignore(numeric_limits<streamsize>::max(), '\n'); 
            return value; // return the valid positive number
        }
    }
}

/*
 * FUNCTION: printPrice
 * Outputs a price value formatted to exactly two decimal places.
 * Example: if price = 5, it will display as 5.00.
 */
void printPrice(double price)
{
    // 'fixed' ensures the number is shown in standard decimal format (not scientific notation).
    // 'setprecision(2)' forces exactly two digits after the decimal point.
    cout << fixed << setprecision(2) << price;
}

/*
 * FUNCTION: selectBike
 * Shows the bike type menu and lets the user choose one.

BikeSelection selectBike()
{
    cout << endl; 
    printDivider(); 
    cout << "  SELECT BIKE TYPE" << endl; // Section title
    printDivider();
    // Display menu options with names and prices 
    cout << "  1. Standard   -  5.00 credits" << endl;
    cout << "  2. Mountain   -  8.50 credits" << endl;
    cout << "  3. Electric   - 12.00 credits" << endl;
    printDivider();

    // Ask user for choice (validated between 1 and NUM_BIKE_TYPES)
    int choice = getValidInt("  Your choice (1-3): ", 1, NUM_BIKE_TYPES);

    BikeSelection bike; // struct to hold chosen bike details

    // Assign name and cost based on the validated choice
    switch (choice)

        bike.typeCost = BIKE_STANDARD;

    case 2:

// FUNTION: selectBikes
// Builds the list of bikes for a booking by repeatedly calling selectBike().
void selectBikes(vector<BikeSelection>& bikes)
{
    bikes.clear();   // ensure we start fresh for a new booking

    // First bike is mandatory
    bikes.push_back(selectBike());

    // Offer additional bikes in a loop
    while (true)
    {
        cout << endl;
        // Ask if user wants another bike
        int again = getValidInt("  Add another bike? (1 = Yes, 2 = No): ", 1, 2);

        if (again == 2) // if No, stop loop
            break;

        // If Yes, add another bike
        bikes.push_back(selectBike());
    }
}

// COST CALCULATION 

// FUNCTION: calculateTotal
// Purpose: Adds up the costs of all bikes, the chosen rental duration, and any accessories to produce the final booking total.
double calculateTotal(
    const vector<BikeSelection>& bikes,  // List of bikes chosen (read-only, passed by const reference)
    double durationCost, // Cost of the rental duration (single value)
    const vector<AccessorySelection>& accessories) // List of accessories chosen (read-only, passed by const reference)
{
    // Start with a running total of 0.0 credits
    double total = 0.0;

    // Loop through each bike in the vector
    for (const BikeSelection& b : bikes)
        total += b.typeCost; // Add the cost of each bike to the running total

    total += durationCost; // Add the rental duration cost once (shared across all bikes)

    // Loop through each accessory in the vector
    for (const AccessorySelection& a : accessories)
        total += a.cost; // Add the cost of each accessory to the running total

    return total; // Return the final computed total as a double
}

// BOOKING SUMMARY

// FUNCTION: printBookingSummary
// Prints a formatted summary of the current booking before checkout. (Shows all bikes, duration, accessories, and the total cost)
void printBookingSummary(
    const vector<BikeSelection>& bikes, // List of bikes chosen (read-only)
    const string& durationName, // Name of rental duration (read-only)
    double durationCost, // Cost of rental duration
    const vector<AccessorySelection>& accessories, // List of accessories chosen (read-only)
    double total) // Final total cost of booking
{
    cout << endl;
    printDivider();
    cout << "  BOOKING SUMMARY" << endl; // Display section title
    printDivider();

    cout << "  Bikes:" << endl; // Heading for bikes section
    for (const BikeSelection& b : bikes) 
    {
        cout << "    - " << b.typeName << "   "; // Print bike name
        printPrice(b.typeCost); // Print bike cost formatted to 2 decimal places
        cout << " credits" << endl; // Label the unit as credits
    }

    cout << "  Duration: " << durationName << "   ";
    printPrice(durationCost); 
    cout << " credits" << endl; 

    if (!accessories.empty()) // Check if any accessories were chosen
    {
        cout << "  Accessories:" << endl; 
        for (const AccessorySelection& a : accessories) // Loop through each accessory
        {
            cout << "    - " << a.name << "   "; 
            printPrice(a.cost); 
            cout << " credits" << endl; 
        }
    }
    else
    {
        cout << "  Accessories: None" << endl; // If no accessories, print "None"
    }

    printDivider();
    cout << "  TOTAL: "; // Heading for total cost
    printPrice(total);
    cout << " credits" << endl; 
    printDivider();
}

// CHECKOUT

// FUNCTION: checkout
// Handles the full checkout process for a booking.
// Ensures the user has enough credits, deducts the total cost, confirms the booking, and returns a completed Booking struct.
Booking checkout(
    const vector<BikeSelection>& bikes, 
    const string& durationName, 
    double durationCost, 
    const vector<AccessorySelection>& accessories, 
    double& creditBalance) 
{
    double total = calculateTotal(bikes, durationCost, accessories); // Step 1: Calculate the total cost of the booking
    printBookingSummary(bikes, durationName, durationCost, accessories, total); // Step 2: Show the booking summary
    // Step 3: Check if the user has enough credits (If not, keep prompting them to add credits until balance >= total)
    while (creditBalance < total)
    {
        cout << endl;
        cout << "  [!] Insufficient credits. You need ";
        printPrice(total - creditBalance); // Show how many more credits are required
        cout << " more credits." << endl;
        cout << "  Please top up to continue." << endl;

        addCredits(creditBalance); // Let the user add credits
    }
    // Step 4: Deduct the total cost from the user’s balance    
    creditBalance -= total;
    // Step 5: Confirm booking to the user
    cout << endl;
    cout << "  Booking confirmed! Thank you." << endl;
    cout << "  Remaining balance: ";
    printPrice(creditBalance);
    cout << " credits" << endl;
    // Step 6: Build a Booking struct with all details    
    Booking newBooking;
    newBooking.bikes = bikes; 
    newBooking.durationName = durationName; 
    newBooking.durationCost = durationCost; 
    newBooking.accessories = accessories;
    newBooking.totalCost = total; 
    return newBooking; // Step 7: Return the completed booking record
}

// RECENT BOOKINGS

// FUNCTION: showRecentBookings
// It displays all bookings made during the current session.
// bookingHistory is passed as a const reference to avoid copying potentially large vectors and to ensure we don’t modify them.

void showRecentBookings(const vector<Booking>& bookingHistory)
{
    cout << endl;
    printDivider();
    cout << "  RECENT BOOKINGS" << endl;
    printDivider();

    // If no bookings exist, show a message and exit
    if (bookingHistory.empty())
    {
        cout << "  No bookings have been made yet." << endl;
        return;
    }

    // Loop through each booking using an index (so we can label them #1, #2, etc.)
    for (size_t i = 0; i < bookingHistory.size(); i++)
    {
        const Booking& b = bookingHistory[i]; // Get the current booking by reference (read-only)

        // Print booking number (index + 1 for human-friendly numbering)
        cout << endl;
        cout << "  Booking #" << (i + 1) << endl;

        cout << "  Bikes:" << endl; // Heading for bikes section
        for (const BikeSelection& bike : b.bikes) // Loop through all bikes in this booking
        {
            cout << "    - " << bike.typeName << "   "; 
            printPrice(bike.typeCost); 
            cout << " credits" << endl; 
        }

        cout << "  Duration: " << b.durationName << "   "; 
        printPrice(b.durationCost);         
        cout << " credits" << endl; 

        if (!b.accessories.empty()) // Check if any accessories were chosen
        {
            cout << "  Accessories:" << endl; 
            for (const AccessorySelection& acc : b.accessories)
            {
                cout << "    - " << acc.name << "   "; 
                printPrice(acc.cost); 
                cout << " credits" << endl;
            }
        }
        else
        {
            cout << "  Accessories: None" << endl; 
        }
        cout << "  Total: ";
        printPrice(b.totalCost); 
        cout << " credits" << endl;
        printDivider();
    }
}

// CREATE BOOKING - Coordinates the full booking workflow (bikes → duration → accessories → checkout) and saves the completed booking to history.

// FUNCTION: createBooking
// creditBalance: must reflect deductions after checkout
// bookingHistory: completed booking is appended here
void createBooking(double& creditBalance, vector<Booking>& bookingHistory)
{
    // Step 1 - choose one or more bikes
    vector<BikeSelection> bikes; // Temporary vector to hold selected bikes
    selectBikes(bikes); // Call function to let user pick bikes

    // Step 2 - choose rental duration (shared across all bikes)
    string durationName; // Variable to store chosen duration name
    double durationCost = 0.0; // Variable to store chosen duration cost
    selectDuration(durationName, durationCost); // Call function to set both values

    // Step 3 - optionally add accessories
    vector<AccessorySelection> accessories; // Temporary vector to hold accessories
    selectAccessories(accessories); // Call function to let user pick accessories

    // Step 4 - confirm and pay (handles insufficient credits internally)
    Booking completed = checkout(bikes, durationName, durationCost, accessories, creditBalance); // Perform checkout, deduct credits, return booking
    bookingHistory.push_back(completed); // Append completed booking to history vector
    cout << endl;
    cout << "  Returning to main menu...." << endl; // Inform user workflow is complete
}

// MAIN 

// FUNCTION: main
// Entry point of the program. Owns all top-level state (credits and booking history) and drives the main application loop until the user chooses to exit.
int main()
{
    printBanner(); // Show the application banner at startup

    double creditBalance = 0.0; // Variable to track the user's credit balance (starts at 0)
    vector<Booking> bookingHistory; // Vector to store all bookings made during the session

    // Ask for an initial credit deposit before showing the menu
    cout << endl;
    cout << "  Welcome to Bike Rental System! Please add some credits to get started." << endl;
    addCredits(creditBalance); // Prompt user to add credits (must be positive)

    // Main application loop - runs until the user chooses Exit
    bool running = true; // true = keep running

    while (running) // Loop continues until running is set to false
    {
        int choice = showMainMenu(creditBalance); // Display main menu and get validated user choice (1–4)

        switch (choice) // Decide action based on user’s choice
        {
        case 1:
            addCredits(creditBalance); // Option 1: Add more credits
            break;

        case 2:
            createBooking(creditBalance, bookingHistory); // Option 2: Start booking workflow (bikes → duration → accessories → checkout)
            break;

        case 3:
            showRecentBookings(bookingHistory); // Option 3: Display all bookings made so far
            break;

        case 4:
            cout << endl;
            cout << "  Thank you for using our service. Goodbye!" << endl;
            running = false; // Option 4: Exit program by setting loop flag to false
            break;
        }
    }
    return 0; // End of program, return 0 to signal successful execution
}
